#
#   # 3. Database + Redis (Docker)
#   cd r3flex-backend && docker compose up -d db redis
#
#   # 4. Apply DB migrations
#   cd r3flex-backend && alembic upgrade head
#
#   # 5. Apply Supabase SQL (one-time, in Supabase SQL editor)
#   #    run files in supabase/migrations/ in order:
#   #      20260421_0001_r3flex_supabase_schema.sql
#   #      20260614_0002_r3flex_security_hardening.sql
